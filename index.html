const express = require('express'); const mongoose = require('mongoose'); const app = express(); app.use(express.json()); // MongoDB Connection (Apna Atlas URI yahan daalein ya Vercel Environment Variable me) const MONGO_URI = process.env.MONGO_URI || 'YOUR_MONGODB_ATLAS_URI_HERE'; mongoose.connect(MONGO_URI) .then(() => console.log('MongoDB Connected Successfully')) .catch(err => console.error('MongoDB Connection Error:', err)); // License Schema const LicenseSchema = new mongoose.Schema({ labId: { type: String, required: true, unique: true }, labName: String, expiryDate: String, // Format: "2027-10-13" working: { type: Boolean, default: true } }); const License = mongoose.model('License', LicenseSchema); // 1. API: Check License Status (Electron app isko call karegi) app.post('/api/check-license', async (req, res) => { try { const { labId } = req.body; if (!labId) return res.status(400).json({ error: 'Lab ID is required' }); let license = await License.findOne({ labId }); // Agar lab pehli baar connect ho rahi hai, toh auto-register kar sakte hain ya block if (!license) { return res.json({ working: false, message: 'Lab not registered on cloud' }); } // Date check: Kya 1 saal cross ho gaya? const today = new Date().toISOString().split('T')[0]; if (today > license.expiryDate) { license.working = false; await license.save(); } res.json({ working: license.working, expiryDate: license.expiryDate, labName: license.labName }); } catch (err) { res.status(500).json({ error: err.message }); } }); // 2. API: Admin Dashboard se Lab Add karne ya Expiry Extend karne ke liye app.post('/api/admin/upsert-license', async (req, res) => { try { const { labId, labName, expiryDate, working, adminSecret } = req.body; // Simple Security Check if (adminSecret !== 'MY_SUPER_SECRET_ADMIN_KEY_123') { return res.status(401).json({ error: 'Unauthorized' }); } const updated = await License.findOneAndUpdate( { labId }, { labName, expiryDate, working }, { upsert: true, new: true } ); res.json({ success: true, updated }); } catch (err) { res.status(500).json({ error: err.message }); } }); const PORT = process.env.PORT || 3000; app.listen(PORT, () => console.log(`Server running on port ${PORT}`));
